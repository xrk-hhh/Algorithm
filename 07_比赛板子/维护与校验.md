# 维护与校验

下面保存本次使用的脚本全文。分别恢复到仓库根目录的 `.maintenance/` 中（该目录为本地工具目录，不提交生成物）。需要 Python 3.9+、ReportLab、Poppler 和 C++14 编译器；脚本中的字体和编译器路径按本机配置调整。

```powershell
python .maintenance/build_original_book.py

pdftoppm -png -r 120 output/pdf/acm_cpp14_contest_cheatsheet.pdf tmp/pdfs/book
```

生成顺序是：按来源选取代码 → 计算 PDF 分页并回写目录页码 → 逐字比对与独立编译 → 运行行为检查。每段完整程序分别编译，不能把多个 main 拼接为同一个程序。修改后应重新渲染检查页码、换栏与长注释换行。

本次结果：18 页 A4，26 个来源比对及 C++14 独立编译通过，173 组行为检查通过；PDF 每页已渲染检查。

```python
from pathlib import Path
import re, json
from reportlab.pdfgen import canvas
from reportlab.lib.pagesizes import A4
from reportlab.lib.units import mm
from reportlab.pdfbase import pdfmetrics
from reportlab.pdfbase.ttfonts import TTFont
from reportlab.lib import colors

ROOT = Path(__file__).resolve().parents[1]
SRC = ROOT / '07_比赛板子/原稿风格板子.md'
OUT = ROOT / 'output/pdf/acm_cpp14_contest_cheatsheet.pdf'
OUT.parent.mkdir(parents=True, exist_ok=True)
pdfmetrics.registerFont(TTFont('CN', 'C:/Windows/Fonts/simhei.ttf'))
pdfmetrics.registerFont(TTFont('Mono', 'C:/Windows/Fonts/consola.ttf'))
W, H = A4
LM, GAP = 13 * mm, 7 * mm
CW = (W - 2 * LM - GAP) / 2
TOP, BOT = H - 19 * mm, 16 * mm
CAP = TOP - BOT
CODE, CLEAD, BODY, BLEAD = 8.5, 10.5, 9.5, 13

def width(s, font, size):
    return sum(pdfmetrics.stringWidth(ch, 'CN' if font == 'CN' or ord(ch) > 127 else 'Mono', size) for ch in s)

def wrap(s, font, size, limit=CW - 6):
    ans, cur = [], ''
    for ch in s.expandtabs(4):
        if cur and width(cur + ch, font, size) > limit:
            ans.append(cur)
            cur = '  ' + ch if font == 'Mono' else ch
        else:
            cur += ch
    ans.append(cur)
    return ans

def token(txt, font, size, lead, sec, comment=False):
    return dict(txt=txt, font=font, size=size, lead=lead, sec=sec, comment=comment)

def tokenize(section, sec):
    result, code = [], False
    for s in section.strip().splitlines():
        if s.startswith(chr(96) * 3):
            code = not code
            if not code:
                result.append(token('', 'CN', 0, 13, sec))
            continue
        if s.startswith('<a ') or s.startswith('<!--'):
            continue
        if code:
            comment = s.lstrip().startswith('//')
            for line in wrap(s, 'Mono', CODE):
                result.append(token(line, 'Mono', CODE, CLEAD, sec, comment))
        elif s.startswith('# '):
            for line in wrap(s[2:], 'CN', 12):
                result.append(token(line, 'CN', 12, 21, sec))
        elif s.strip():
            s = re.sub(r'\[([^\]]+)\]\([^)]*\)', r'\1', s)
            s = s.replace(chr(96), '').replace('**', '')
            s = s.translate(str.maketrans({'³': '^3', '²': '^2', 'α': 'alpha'}))
            for line in wrap(s, 'CN', BODY):
                result.append(token(line, 'CN', BODY, BLEAD, sec))
            result.append(token('', 'CN', 0, 8, sec))
    return result

source = SRC.read_text(encoding='utf-8')
sections = [s for s in source.split('<!-- page -->') if s.strip()]
titles = [s.strip().splitlines()[0].removeprefix('# ') for s in sections]
cols = [[]]
heights = [0]
starts = [1]
for sec, section in enumerate(sections[1:], 1):
    ts = tokenize(section, sec)
    # Keep the title, explanation, and the start of the source together.
    first_code = next((i for i, t in enumerate(ts) if t['font'] == 'Mono'), len(ts))
    reserve = sum(t['lead'] for t in ts[:first_code + 8])
    if heights[-1] and heights[-1] + reserve > CAP:
        cols.append([])
        heights.append(0)
    starts.append(2 + (len(cols) - 1) // 2)
    for t in ts:
        if heights[-1] + t['lead'] > CAP and cols[-1]:
            cols.append([])
            heights.append(0)
        cols[-1].append(t)
        heights[-1] += t['lead']
pages = [cols[i:i + 2] for i in range(0, len(cols), 2)]
content_total = 1 + len(pages)
total = content_total + content_total % 2
first = sections[0]
for idx in range(1, len(sections)):
    first = re.sub(
        r'(?m)^- \[([^\]]+)\]\(#part' + f'{idx:02d}' + r'\)',
        lambda m: '- [' + re.sub(r' · 第 \d+ 页$', '', m.group(1)) + f' · 第 {starts[idx]} 页](#part{idx:02d})',
        first, count=1)
sections[0] = first
SRC.write_text('<!-- page -->' + '<!-- page -->'.join(sections), encoding='utf-8')
report = dict(pages=total, content_pages=content_total, starts=starts,
              column_heights=[round(h, 2) for h in heights])
(ROOT / '.maintenance/layout_report.json').write_text(json.dumps(report, indent=2), encoding='utf-8')
for col, h in zip(cols, heights):
    assert h <= CAP + .01
    for t in col:
        assert not t['size'] or width(t['txt'], t['font'], t['size']) <= CW

c = canvas.Canvas(str(OUT), pagesize=A4)
c.setTitle('C++14 ACM 原稿风格板子')
c.setAuthor('xrk-hhh')
c.setViewerPreference('Duplex', 'DuplexFlipLongEdge')

def draw_text(txt, font, size, x, y):
    for ch in txt:
        f = 'CN' if font == 'CN' or ord(ch) > 127 else 'Mono'
        c.setFont(f, size)
        c.drawString(x, y, ch)
        x += pdfmetrics.stringWidth(ch, f, size)

def chrome(page):
    c.setStrokeColor(colors.HexColor('#9aa3aa'))
    c.setLineWidth(.4)
    c.line(LM, H - 12 * mm, W - LM, H - 12 * mm)
    c.setFillColor(colors.HexColor('#535c63'))
    c.setFont('CN', 7.5)
    c.drawString(LM, H - 9.5 * mm, 'C++14 ACM 算法板子')
    c.drawString(LM, 9 * mm, 'xrk-hhh / Algorithm')
    c.drawRightString(W - LM, 9 * mm, f'{page:02d} / {total:02d}')

# A full-width directory is the only content on page one.
chrome(1)
c.bookmarkPage('directory')
c.addOutlineEntry('目录', 'directory', 0)
c.setFillColor(colors.black)
c.setFont('CN', 19)
c.drawString(LM, TOP - 3, '目录')
c.setFont('CN', 10)
c.drawRightString(W - LM, TOP - 3, 'C++14 ACM 原稿风格板子')
y = TOP - 39
for i, title in enumerate(titles[1:], 1):
    title = re.sub(r'^\d+\s+', '', title)
    c.setFont('CN', 10.5)
    c.setFillColor(colors.black)
    label = f'{i:02d}  {title}'
    c.drawString(LM + 3, y, label)
    page_label = str(starts[i])
    c.drawRightString(W - LM - 3, y, page_label)
    a = LM + 10 + width(label, 'CN', 10.5)
    b = W - LM - 22
    if a < b:
        c.setStrokeColor(colors.HexColor('#c7ccd0'))
        c.setDash(1, 3)
        c.line(a, y + 2, b, y + 2)
        c.setDash()
    c.linkRect('', f's{i}', (LM, y - 4, W - LM, y + 12), relative=0, thickness=0)
    y -= 24
assert y > BOT
c.showPage()
seen = set()
for page_idx, pg in enumerate(pages, 2):
    chrome(page_idx)
    for j, col in enumerate(pg):
        x, y = LM + j * (CW + GAP), TOP
        if col and col[0]['size'] < 12:
            c.setFillColor(colors.HexColor('#6a737a'))
            draw_text('续 · ' + titles[col[0]['sec']], 'CN', 7, x, TOP + 10)
        for t in col:
            if t['size'] >= 12 and t['sec'] not in seen:
                sec = t['sec']
                c.bookmarkHorizontalAbsolute(f's{sec}', y + 15)
                c.addOutlineEntry(titles[sec], f's{sec}', 0)
                seen.add(sec)
            if t['font'] == 'Mono':
                c.setFillColor(colors.HexColor('#f3f4f5'))
                c.rect(x - 3, y - 2.5, CW + 3, t['lead'], fill=1, stroke=0)
            c.setFillColor(colors.HexColor('#45564d') if t['comment'] else colors.black)
            if t['size']:
                draw_text(t['txt'], t['font'], t['size'], x, y)
            y -= t['lead']
    c.showPage()
if total > content_total:
    c.showPage()
c.save()
print(f'PDF {total} pages ({content_total} content); {CODE}pt code; original source unchanged')
```

```python
from pathlib import Path
import re, json, subprocess, concurrent.futures, hashlib
ROOT=Path(__file__).resolve().parents[1]
manifest=json.loads((ROOT/'.maintenance/book_sources.json').read_text(encoding='utf-8'))
md=(ROOT/'07_比赛板子/原稿风格板子.md').read_text(encoding='utf-8')
f=chr(96)*3
blocks=re.findall(f+'cpp\n(.*?)\n'+f,md,re.S)
assert len(blocks)==len(manifest)
tmp=ROOT/'.maintenance/book_units'
tmp.mkdir(exist_ok=True)
def check(it):
    i,code=it
    unit=manifest[i-1]
    raw=(ROOT/unit['source']).read_text(encoding='utf-8-sig')
    lines=raw.splitlines(keepends=True)
    if len(lines)>=2 and lines[0].startswith('// 原始稿：') and lines[1].startswith('// 原文行号：'):
        raw=''.join(lines[2:])
    assert code==raw,(i,'source differs')
    src=tmp/f'{i:02d}.cpp'
    src.write_text(code,encoding='utf-8')
    z=subprocess.run(['D:/tdm64-gcc/bin/g++.exe','-std=c++14','-pedantic-errors','-O2',
                      str(src),'-o',str(tmp/f'{i:02d}.exe')],
                      capture_output=True,text=True,encoding='utf-8',errors='replace')
    assert z.returncode==0,(i,z.stderr)
    return dict(id=i,title=unit['title'],source=unit['source'],same_source=True,
                compile=True,sha256=hashlib.sha256(code.encode('utf-8')).hexdigest())
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as pool:
    result=list(pool.map(check,enumerate(blocks,1)))
(ROOT/'.maintenance/book_check.json').write_text(json.dumps(result,ensure_ascii=False,indent=2),encoding='utf-8')
print(f'{len(result)} exact source comparisons and C++14 independent compilations passed')
```
