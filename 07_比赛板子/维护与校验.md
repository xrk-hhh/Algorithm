# 维护与校验

需要 Python 3.9+、ReportLab、Matplotlib、fontTools、pypdf、Poppler 和 GNU C++14。脚本中的字体及编译器路径可按环境修改。下文完整收录数据与脚本，分别恢复至仓库根目录的 .maintenance/。

python .maintenance/build_contest_book.py

python .maintenance/check_contest_book.py

pdftoppm -png -r 120 output/pdf/acm_cpp14_contest_cheatsheet.pdf tmp/pdfs/final/page

book_topics.txt 的字段为“知识点键 | 大类/子类 | 方法名 | 题目描述 | 变式例题 | 关键说明”；book_formulas.json 保存各知识点的 TeX 公式，addition_sources.json 保存扩展来源。同一大类内，同名子类使用相同排序编号。构建脚本读取源码，排版脚本计算双栏分页并回写目录实际页码；行为检查按来源映射，不依赖当前目录编号。

当前结果：8 个大类、49 个子类、179 个知识点，103 条公式；179 次源码比对和 C++14 编译通过，403 组行为用例通过。PDF 为 145 页 A4，双面 73 张；代码 8.5pt、正文 9.5pt，前 4 页为目录。PDF、可执行文件及缓存属于本地输出，不提交仓库。

公式在 PDF 中转为矢量路径，分式、上下标与求和号使用数学排版；宽公式缩放后基础字号不得小于 8pt。公式过宽时应拆为多行。PDF 公式为路径，检索与复制公式请使用 Markdown。代码注释中的特殊字符使用后备字体，避免缺字；打印换行不改变 Markdown 中的代码。

def source_text(unit):
    raw=(root/unit['source']).read_text(encoding='utf-8-sig')
    if unit['source'].endswith('.md'):
        blocks=re.findall(chr(96)*3+r'cpp\n(.*?)\n'+chr(96)*3,raw,re.S)
        raw=next(block for block in blocks if ('int josephus(' in block if unit['key']=='约瑟夫幸存者递推' else '#include' in block and 'main' in block))+'\n'
    lines=raw.splitlines(keepends=True)
    if len(lines)>=2 and lines[0].startswith('// 原始稿：') and lines[1].startswith('// 原文行号：'):
        raw=''.join(lines[2:])
    if unit['kind']=='snippet':

        brace=raw.index('{',start)
        depth=1
        end=brace+1
        while depth:
            depth+=(raw[end]=='{')-(raw[end]=='}')

        assert raw[end]==';'
        raw=raw[start:end+1]+'\n'
    return raw

def wrap(s,font,size,limit=CW-7):
    # Preserve indentation; prefer a word boundary for prose and long comments.
    s=s.expandtabs(4)
    ans=[]
    indent=' '*(len(s)-len(s.lstrip(' ')))
    continuation=(indent+'  ') if font=='Mono' else ''
    while width(s,font,size)>limit:

        for ch in s:
            if width(cur+ch,font,size)>limit:break
            cur+=ch
        cut=len(cur)

        safe=max(s.rfind(' ',len(cur)//2,len(cur)),s.rfind('，',len(cur)//2,len(cur)))

    return ans
